Architecture

How It Works

From your microphone to on-device neural processing and back in ~1 second.

1
Audio Capture & 16kHz PCM
Microphone voice is captured and resampled directly into standard 16kHz 16-bit Mono WAV in the browser.
2
Anycast Edge
HTTPS POST request reaches Cloudflare's global edge network nearest to the client (sub-10ms).
3
QUIC Tunnel
Cloudflare multiplexes the request through an encrypted QUIC tunnel directly into the phone's cloudflared daemon.
4
On-Device AI
Alibaba SenseVoice-Small engine executes non-autoregressive neural decode in phone RAM using 4 ARM CPU threads, returning structured text.

Universal drop-in snippet for Google Colab, Jupyter Notebooks, FastAPI, Flask, or backend services supporting all 5 AI modes: 

import requests, json

BASE_URL = "https://black-term-8c36.botmaker583-55e.workers.dev"

# 1. Speech-to-Text (Alibaba SenseVoice-Small)
def transcribe(audio_path: str):
    with open(audio_path, "rb") as f:
        res = requests.post(f"{BASE_URL}/inference", files={"file": f}, data={"temperature": "0.0"})
        return res.json()

# 2. Token Streaming Chat (Qwen 2.5 SLM)
def chat_stream(prompt: str):
    payload = {"messages": [{"role": "user", "content": prompt}], "stream": True}
    res = requests.post(f"{BASE_URL}/v1/chat/completions", json=payload, headers={"Accept": "text/event-stream"}, stream=True)
    for line in res.iter_lines(decode_unicode=True):
        if line and line.startswith("data: ") and "[DONE]" not in line:
            chunk = json.loads(line[6:])
            yield chunk["choices"][0]["delta"].get("content", "")

# 3. Text-to-Speech (On-Device TTS)
def text_to_speech(text: str, out_file="speech.wav"):
    res = requests.post(f"{BASE_URL}/v1/audio/speech", json={"input": text, "speed": 1.0})
    with open(out_file, "wb") as f:
        f.write(res.content)

# 4. Dense Vector Embeddings (896 Dimensions)
def embed_text(text: str):
    res = requests.post(f"{BASE_URL}/v1/embeddings", json={"input": text})
    return res.json()["data"][0]["embedding"]

# 5. Real-Time Hardware Telemetry
def get_telemetry():
    return requests.get(f"{BASE_URL}/telemetry").json()

Native Fetch with zero npm dependencies — works in Node.js (v18+), React, Vue, Next.js, and Browsers: 

const BASE_URL = "https://black-term-8c36.botmaker583-55e.workers.dev";

// 1. Speech-to-Text (STT)
async function transcribe(audioBlob) {
  const fd = new FormData();
  fd.append("file", audioBlob, "audio.wav");
  const res = await fetch(`${BASE_URL}/inference`, { method: "POST", body: fd });
  return await res.json();
}

// 2. Token-by-Token Streaming Chat (Qwen 2.5 SLM)
async function streamChat(prompt, onToken) {
  const res = await fetch(`${BASE_URL}/v1/chat/completions`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ messages: [{ role: "user", content: prompt }], stream: true })
  });
  const reader = res.body.getReader();
  const decoder = new TextDecoder();
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    const text = decoder.decode(value);
    for (const line of text.split("\n")) {
      if (line.startsWith("data: ") && !line.includes("[DONE]")) {
        const delta = JSON.parse(line.slice(6))?.choices?.[0]?.delta?.content;
        if (delta) onToken(delta);
      }
    }
  }
}

// 3. Text-to-Speech (TTS)
async function textToSpeech(text) {
  const res = await fetch(`${BASE_URL}/v1/audio/speech`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ input: text })
  });
  return await res.blob();
}

// 4. Vector Embeddings
async function getEmbedding(text) {
  const res = await fetch(`${BASE_URL}/v1/embeddings`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ input: text })
  });
  const data = await res.json();
  return data.data[0].embedding;
}

// 5. Hardware Telemetry
async function fetchTelemetry() {
  return await fetch(`${BASE_URL}/telemetry`).then(r => r.json());
}
